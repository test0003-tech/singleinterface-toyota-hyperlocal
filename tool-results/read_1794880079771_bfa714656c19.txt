   896→function MediaSpendForm({

   900→  liveStores,
   901→  presetCode,
   902→  presetMonth,
   903→  onSubmit,
   904→}: {
   905→  open: boolean
   906→  onOpenChange: (v: boolean) => void
   907→  editing: MediaSpend | null
   908→  liveStores: LiveSpotlightStore[]
   909→  presetCode: string | null
   910→  presetMonth: string | null

   913→  const [recordCode, setRecordCode] = useState('')
   914→  const [month, setMonth] = useState(currentMonth())
   915→  const [metaAllocated, setMetaAllocated] = useState('')

   918→  const [googleUtilized, setGoogleUtilized] = useState('')
   919→  const [notes, setNotes] = useState('')
   920→  const [submitting, setSubmitting] = useState(false)
   921→  const [error, setError] = useState<string | null>(null)
   922→  const [storePickerOpen, setStorePickerOpen] = useState(false)
   923→  const [manualEntry, setManualEntry] = useState(false)
   924→
   925→  useEffect(() => {
   926→    if (open) {
   927→      setRecordCode(editing?.recordCode ?? presetCode ?? '')
   928→      setMonth(editing?.month ?? presetMonth ?? currentMonth())
   929→      setMetaAllocated(editing ? String(editing.metaAllocated) : '')
   930→      setMetaUtilized(editing ? String(editing.metaUtilized) : '')
   931→      setGoogleAllocated(editing ? String(editing.googleAllocated) : '')

   938→        !editing && !presetCode

   943→  }, [open, editing, presetCode, presetMonth, liveStores])

   955→
   956→  // Live preview of pending = allocated - utilized (excludes prev. month balance

   959→  const googlePreview = (parseFloat(googleAllocated) || 0) - (parseFloat(googleUtilized) || 0)

   988→  return (

   990→      <DialogContent className="sm:max-w-xl">
   991→        <DialogHeader>
   992→          <DialogTitle>

   999→          </DialogDescription>

  1246→function downloadCSV(rows: MediaSpend[], filename: string) {
  1247→  const headers = [
  1248→    'Location Type',
  1249→    'Location Code',
  1250→    'Month',
  1251→    'Meta Allocated',
  1252→    'Meta Utilized',
  1253→    'Meta Pending in Wallet',
  1254→    'Google Allocated',
  1255→    'Google Utilized',
  1256→    'Google Pending in Wallet',
  1257→    'Total Allocated',
  1258→    'Total Utilized',
  1259→    'Total Pending in Wallet',
  1260→    'Notes',
  1261→  ]
  1262→  const escape = (v: unknown) => {
  1263→    if (v === null || v === undefined) return ''
  1264→    const s = String(v)
  1265→    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`
  1266→    return s
  1267→  }
  1268→  const csv = [
  1269→    headers.join(','),
  1270→    ...rows.map((r) =>
  1271→      [
  1272→        r.recordType,
  1273→        r.recordCode,
  1274→        r.month,
  1275→        r.metaAllocated,
  1276→        r.metaUtilized,
  1277→        r.metaPendingInWallet,
  1278→        r.googleAllocated,
  1279→        r.googleUtilized,
  1280→        r.googlePendingInWallet,
  1281→        r.allocated,
  1282→        r.utilized,
  1283→        r.pendingInWallet,
  1284→        r.notes,
  1285→      ]
  1286→        .map(escape)
  1287→        .join(',')
  1288→    ),
  1289→  ].join('\n')
  1290→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' })
  1291→  const url = URL.createObjectURL(blob)
  1292→  const a = document.createElement('a')
  1293→  a.href = url
  1294→  a.download = filename
  1295→  a.click()
  1296→  URL.revokeObjectURL(url)
  1297→}
