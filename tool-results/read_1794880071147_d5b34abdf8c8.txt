   101→async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   102→  const res = await fetch(url, init)
   103→  if (!res.ok) {
   104→    const body = await res.json().catch(() => null)
   105→    throw new Error(body?.error || `Request failed (${res.status})`)
   106→  }
   107→  return res.json() as Promise<T>
   108→}
   109→
   110→function linkRecords(siId: string, spotlightCode: string | null) {
   111→  return fetchJson<{ success: boolean }>('/api/location-mapping/link', {
   112→    method: 'POST',
   113→    headers: { 'Content-Type': 'application/json' },
   114→    body: JSON.stringify({ siId, spotlightCode }),
   115→  })
   116→}

   255→export function LocationMappingPage() {
   256→  const { toast } = useToast()
   257→  const [data, setData] = useState<LocationMappingData | null>(null)
   258→  const [error, setError] = useState<string | null>(null)
   259→  const [refreshing, setRefreshing] = useState(false)
   260→  const [minScore, setMinScore] = useState(50)
   261→  const [tab, setTab] = useState<TabId>('mapped')
   262→  const [query, setQuery] = useState('')
   263→  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE)
   264→  const [busySiId, setBusySiId] = useState<string | null>(null)
   265→  const [autoLinkProgress, setAutoLinkProgress] = useState<{ done: number; total: number } | null>(
   266→    null
   267→  )
   268→  const hasDataRef = useRef(false)
   269→
   270→  const fetchData = useCallback(
   271→    async (score: number) => {
   272→      setRefreshing(true)
   273→      setError(null)
   274→      try {
   275→        const json = await fetchJson<LocationMappingData>(
   276→          `/api/location-mapping?minScore=${score}`
   277→        )
   278→        setData(json)
   279→        hasDataRef.current = true
   280→      } catch (e) {
   281→        const msg = e instanceof Error ? e.message : 'Failed to load location mapping'
   282→        setError(msg)
   283→        if (hasDataRef.current) {
   284→          toast({
   285→            title: 'Refresh failed',
   286→            description: msg,
   287→            variant: 'destructive',
   288→          })
   289→        }
   290→      } finally {
   291→        setRefreshing(false)
   292→      }
   293→    },
   294→    [toast]
   295→  )
   296→
   297→  useEffect(() => {
   298→    fetchData(minScore)
   299→  }, [fetchData, minScore])
   300→
   301→  // Reset pagination when the view changes
   302→  useEffect(() => {
   303→    setVisibleCount(PAGE_SIZE)
   304→  }, [tab, query])
   305→
   306→  // ---- Derived data ----
   307→
   308→  const mappedRows = useMemo(() => {
   309→    if (!data) return []
   310→    return data.mappings
   311→      .filter((m) => m.matchType !== 'none' && m.spotlightCode)
   312→      .sort((a, b) => {
   313→        if (a.matchType !== b.matchType) return a.matchType === 'manual' ? -1 : 1
   314→        return b.matchScore - a.matchScore
   315→      })
   316→  }, [data])
   317→
   318→  const unmatchedSiRows = useMemo(
   319→    () => (data ? data.mappings.filter((m) => m.matchType === 'none') : []),
   320→    [data]
   321→  )
   322→
   323→  const unmatchedSpotRows = data?.unmatchedSpotlight ?? []
   324→
   325→  const q = query.trim().toLowerCase()
   326→
   327→  const filteredMapped = useMemo(() => {
   328→    if (!q) return mappedRows
   329→    return mappedRows.filter((r) =>
   330→      [
   331→        r.siCode,
   332→        r.siClientCode,
   333→        r.siName,
   334→        r.siCity,
   335→        r.siState,
   336→        r.spotlightCode,
   337→        r.spotlightName,
   338→        r.spotlightCity,
   339→        r.spotlightState,
   340→      ].some((t) => !!t && t.toLowerCase().includes(q))
   341→    )
   342→  }, [mappedRows, q])
   343→
   344→  const filteredUnmatchedSi = useMemo(() => {
   345→    if (!q) return unmatchedSiRows
   346→    return unmatchedSiRows.filter((r) =>
   347→      [r.siCode, r.siClientCode, r.siName, r.siCity, r.siState, r.siCircle].some(
   348→        (t) => !!t && t.toLowerCase().includes(q)
   349→      )
   350→    )
   351→  }, [unmatchedSiRows, q])
   352→
   353→  const filteredUnmatchedSpot = useMemo(() => {
   354→    if (!q) return unmatchedSpotRows
   355→    return unmatchedSpotRows.filter((r) =>
   356→      [r.clientCode, r.name, r.city, r.state, r.region, r.domain].some(
   357→        (t) => !!t && t.toLowerCase().includes(q)
   358→      )
   359→    )
   360→  }, [unmatchedSpotRows, q])
   361→
   362→  // ---- Actions ----
   363→
   364→  async function handleLink(row: MappingRow, spotlightCode: string | null) {
   365→    setBusySiId(row.siId)
   366→    try {
   367→      await linkRecords(row.siId, spotlightCode)
   368→      toast({
   369→        title: spotlightCode ? 'Records linked' : 'Link removed',
   370→        description: spotlightCode
   371→          ? `SI ${row.siCode} is now linked to Spotlight ${spotlightCode}`
   372→          : `SI ${row.siCode} was unlinked from Spotlight`,
   373→      })
   374→      await fetchData(minScore)
   375→    } catch (e) {
   376→      toast({
   377→        title: spotlightCode ? 'Link failed' : 'Unlink failed',
   378→        description: e instanceof Error ? e.message : 'Something went wrong',
   379→        variant: 'destructive',
   380→      })
   381→    } finally {
   382→      setBusySiId(null)
   383→    }
   384→  }
   385→
   386→  async function handleAutoLinkAll() {
   387→    if (!data || autoLinkProgress) return
   388→    const targets = data.mappings.filter((m) => m.matchType === 'auto' && m.spotlightCode)
   389→    if (targets.length === 0) {
   390→      toast({
   391→        title: 'Nothing to auto-link',
   392→        description: 'No auto-matched pairs above the current score threshold.',
   393→      })
   394→      return
   395→    }
   396→    setAutoLinkProgress({ done: 0, total: targets.length })
   397→    let linked = 0
   398→    let failed = 0
   399→    for (let i = 0; i < targets.length; i++) {
   400→      try {
   401→        await linkRecords(targets[i].siId, targets[i].spotlightCode)
   402→        linked++
   403→      } catch {
   404→        failed++
   405→      }
   406→      setAutoLinkProgress({ done: i + 1, total: targets.length })
   407→    }
   408→    setAutoLinkProgress(null)
   409→    toast({
   410→      title: 'Auto-link complete',
   411→      description: `${linked} pair${linked === 1 ? '' : 's'} linked${
   412→        failed > 0 ? `, ${failed} failed` : ''
   413→      }.`,
   414→      variant: failed > 0 && linked === 0 ? 'destructive' : undefined,
   415→    })
   416→    await fetchData(minScore)
   417→  }
   418→
   419→  // ---- Render ----
   420→
   421→  if (!data) {
   422→    if (error) {
   423→      return (
   424→        <Card className="border-destructive/30">
   425→          <CardContent className="pt-6 flex items-start gap-3">
   426→            <AlertCircle className="h-5 w-5 text-destructive mt-0.5 shrink-0" />
   427→            <div>
   428→              <p className="text-sm font-medium text-destructive">
   429→                Failed to load location mapping
   430→              </p>
   431→              <p className="text-xs text-muted-foreground mt-0.5">{error}</p>
   432→              <Button variant="outline" size="sm" className="mt-3" onClick={() => fetchData(minScore)}>
   433→                <RefreshCw className="mr-2 h-4 w-4" />
   434→                Retry
   435→              </Button>
   436→            </div>
   437→          </CardContent>
   438→        </Card>
   439→      )
   440→    }
   441→    return (
   442→      <div className="space-y-5">
   443→        <div className="space-y-2">
   444→          <Skeleton className="h-7 w-56" />
   445→          <Skeleton className="h-4 w-96 max-w-full" />
   446→        </div>
   447→        <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-6 gap-3">
   448→          {Array.from({ length: 6 }).map((_, i) => (
   449→            <Card key={i}>
   450→              <CardContent className="pt-4 pb-4 space-y-2">
   451→                <Skeleton className="h-3 w-20" />
   452→                <Skeleton className="h-7 w-14" />
   453→              </CardContent>
   454→            </Card>
   455→          ))}
   456→        </div>
   457→        <div className="rounded-lg border bg-card overflow-hidden">
   458→          <div className="p-3 flex items-center gap-2 border-b">
   459→            <Skeleton className="h-9 w-64" />
   460→            <Skeleton className="h-9 w-40" />
   461→          </div>
   462→          <div className="p-4 space-y-3">
   463→            {Array.from({ length: 6 }).map((_, i) => (
   464→              <div key={i} className="flex items-center gap-4">
   465→                <Skeleton className="h-12 w-1/3" />
   466→                <Skeleton className="h-5 w-6" />
   467→                <Skeleton className="h-12 w-1/3" />
   468→                <Skeleton className="h-5 w-20" />
   469→              </div>
   470→            ))}
   471→          </div>
   472→        </div>
   473→      </div>
   474→    )
   475→  }
   476→
   477→  const summary = data.summary
   478→  const actionsDisabled = refreshing || !!autoLinkProgress
   479→  const autoLinkCount = summary.autoMatched
   480→
   481→  return (
   482→    <div className="space-y-5">
   483→      {/* Header */}
   484→      <div className="flex flex-col gap-4 lg:flex-row lg:items-center lg:justify-between">
   485→        <div>
   486→          <h2 className="text-xl font-semibold tracking-tight flex items-center gap-2">
   487→            <Link2 className="h-5 w-5 text-primary" />
   488→            Location Mapping
   489→          </h2>
   490→          <p className="text-sm text-muted-foreground mt-0.5">
   491→            SI ↔ Spotlight record mapping — auto-matched by dealer name, city, and state
   492→          </p>
   493→        </div>
   494→        <div className="flex flex-wrap items-center gap-2">
   495→          <div className="flex items-center gap-2">
   496→            <Label htmlFor="min-score" className="text-xs text-muted-foreground whitespace-nowrap">
   497→              Min Score
   498→            </Label>
   499→            <Select
   500→              value={String(minScore)}
   501→              onValueChange={(v) => setMinScore(Number(v))}
   502→              disabled={actionsDisabled}
   503→            >
   504→              <SelectTrigger id="min-score" className="w-[88px] h-9">
   505→                <SelectValue />
   506→              </SelectTrigger>
   507→              <SelectContent>
   508→                {MIN_SCORE_OPTIONS.map((s) => (
   509→                  <SelectItem key={s} value={String(s)}>
   510→                    {s}%
   511→                  </SelectItem>
   512→                ))}
   513→              </SelectContent>
   514→            </Select>
   515→          </div>
   516→          <Button
   517→            size="sm"
   518→            onClick={handleAutoLinkAll}
   519→            disabled={actionsDisabled || autoLinkCount === 0}
   520→            title="Confirm all auto-matched pairs as manual links"
   521→          >
   522→            {autoLinkProgress ? (
   523→              <>
   524→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   525→                Linking {autoLinkProgress.done}/{autoLinkProgress.total}…
   526→              </>
   527→            ) : (
   528→              <>
   529→                <Zap className="mr-2 h-4 w-4" />
   530→                Auto-Link All{autoLinkCount > 0 ? ` (${autoLinkCount})` : ''}
   531→              </>
   532→            )}
   533→          </Button>
   534→          <Button
   535→            variant="outline"
   536→            size="sm"
   537→            onClick={() => fetchData(minScore)}
   538→            disabled={actionsDisabled}
   539→          >
   540→            <RefreshCw className={`mr-2 h-4 w-4 ${refreshing ? 'animate-spin' : ''}`} />
   541→            Refresh
   542→          </Button>
   543→        </div>
   544→      </div>
   545→
   546→      {/* Summary cards */}
   547→      <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-6 gap-3">
   548→        <SummaryCard
   549→          label="Total SI"
   550→          value={summary.totalSI}
   551→          icon={<Building2 className="h-3.5 w-3.5" />}
   552→        />
   553→        <SummaryCard
   554→          label="Total Spotlight"
   555→          value={summary.totalSpotlight}
   556→          icon={<MapPin className="h-3.5 w-3.5" />}
   557→        />
   558→        <SummaryCard
   559→          label="Matched"
   560→          value={summary.matched}
   561→          valueClass="text-emerald-600"
   562→          icon={<CheckCircle2 className="h-3.5 w-3.5" />}
   563→        />
   564→        <SummaryCard
   565→          label="Manually Linked"
   566→          value={summary.manuallyLinked}
   567→          valueClass="text-primary"
   568→          icon={<Link2 className="h-3.5 w-3.5" />}
   569→        />
   570→        <SummaryCard
   571→          label="Auto-Matched"
   572→          value={summary.autoMatched}
   573→          valueClass="text-amber-600"
   574→          icon={<Zap className="h-3.5 w-3.5" />}
   575→        />
   576→        <SummaryCard
   577→          label="Match Rate"
   578→          value={`${summary.matchRate}%`}
   579→          valueClass="text-emerald-600"
   580→          icon={<TrendingUp className="h-3.5 w-3.5" />}
   581→        />
   582→      </div>
   583→
   584→      {/* Tabs: mapped / unmatched SI / unmatched Spotlight */}
   585→      <Tabs value={tab} onValueChange={(v) => setTab(v as TabId)}>
   586→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   587→          <TabsList className="flex-wrap h-auto">
   588→            <TabsTrigger value="mapped" className="gap-1.5">
   589→              <Link2 className="h-3.5 w-3.5" />
   590→              All Mapped
   591→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] font-semibold">
   592→                {summary.matched}
   593→              </Badge>
   594→            </TabsTrigger>
   595→            <TabsTrigger value="unmatched-si" className="gap-1.5">
   596→              <Building2 className="h-3.5 w-3.5" />
   597→              Unmatched SI
   598→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] font-semibold">
   599→                {summary.unmatchedSI}
   600→              </Badge>
   601→            </TabsTrigger>

   603→              <MapPin className="h-3.5 w-3.5" />

   605→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] font-semibold">
   606→                {summary.unmatchedSpotlight}
